use serde::Deserialize;
use std::error::Error;

#[derive(Debug, Deserialize, Clone)]
pub struct Config {
    pub system: System,
    pub integrations: Integrations,
    #[serde(default)]
    pub mqtt: Mqtt,
    #[serde(default)]
    pub http: Vec<Http>,
    #[serde(default)]
    pub helper: Helper,
}

impl Config {
    pub fn load_config(path: &str) -> Result<Self, Box<dyn Error + Send + Sync>> {
        let cfg_str = std::fs::read_to_string(path)?;
        let config: Config = toml::from_str(&cfg_str)?;

        let has_integration =
            config.integrations.mqtt.is_some() || config.integrations.modbus_rtu.is_some();
        let has_helper = !config.helper.input_boolean.is_empty();

        if !has_integration && !has_helper {
            return Err(format!(
                "Конфиг {path} пуст: нет ни интеграций (mqtt/modbus_rtu), \
                 ни helpers (input_boolean). Движку не с чем работать."
            )
            .into());
        }

        Ok(config)
    }
}

/// Пульт управления EntityManager.
/// Клонируется, раздаётся модулям.
#[derive(Clone)]
pub struct EntityHandle {
    tx_to_router: UnboundedSender<SystemMessage>,
}

impl EntityHandle {
    pub fn new(tx_to_router: UnboundedSender<SystemMessage>) -> Self {
        Self { tx_to_router }
    }

    /// Блокирующий метод для синхронного кода Lua.
    /// Он отправляет запрос в Роутер и останавливает поток Lua до получения ответа от Ядра.
    pub  fn get_state_blocking(&self, entity_id: &str) -> Option<Entity> {
        // Создаем одноразовый канал для получения ответа
        let (reply_tx, reply_rx) = oneshot::channel();

        // Упаковываем запрос в сообщение для Роутера
        let msg = SystemMessage::GetEntityState { 
            entity_id: entity_id.to_string(), 
            reply: reply_tx };

        // Отправляем сообщение на центральный Роутер
        if self.tx_to_router.send(msg).is_err() {
            return None;
        };

        // БЛОКИРУЕМ текущий поток ОС (в котором крутится Lua) до тех пор,
        // пока Ядро не пришлет нам ответ в reply_rx
        reply_rx.blocking_recv().ok()?
    }

// ./crates/ent_core/src/manager.rs
use std::collections::HashMap;
use std::{println, thread, eprintln};
use tokio::sync::mpsc::{UnboundedReceiver, UnboundedSender};

// Импортируем всё необходимое из единого центра типов
use ent_types::{SystemMessage, Entity, EntityState, FieldUpdate};

pub struct EntityManager {
    // приемник от Роутера
    rx: UnboundedReceiver<SystemMessage>,
    // передатчик в Роутер (чтобы кричать о событиях)
    tx_to_router: UnboundedSender<SystemMessage>,
    
    // внутренняя база данных сущностей
    pub entities: HashMap<String, Entity>,
}

impl EntityManager {
    /// Спавнит менеджер в отдельном потоке ОС
    /// Теперь на вход принимаем наш rx_for_core и клон tx_in из main.rs
    pub fn spawn(
        rx: UnboundedReceiver<SystemMessage>,
        tx_to_router: UnboundedSender<SystemMessage>,
    ) {
        let mut manager = Self {
            rx,
            tx_to_router,
            entities: HashMap::new(),
        };

        // Оставляем менеджер в собственном потоке ОС
        thread::Builder::new()
            .name("[entcore] entity-manager".into())
            .spawn(move || {
                // Создаем локальный микро-рантайм Tokio для асинхронного цикла чтения
                let rt = tokio::runtime::Builder::new_current_thread()
                    .enable_all()
                    .build()
                    .expect("Не удалось запустить рантайм для EntityManager");

                rt.block_on(manager.run());
            })
            .expect("Не удалось запустить поток EntityManager");
    }

    /// Главный асинхронный цикл обработки данных ядра
    async fn run(&mut self) {
        println!("[Ядро] Менеджер сущностей запущен.");

        while let Some(msg) = self.rx.recv().await {
            match msg {
                // 1. РЕГИСТРАЦИЯ СУЩНОСТИ
                SystemMessage::RegisterEntity(entity) => {
                    let id = entity.id.clone();
                    if self.entities.contains_key(&id) {
                        println!("[Ядро] Сущность с id = {} уже существует", id);
                        continue;
                    }

                    self.entities.insert(id.clone(), entity);
                    println!("[Ядро] Сущность с id = {} успешно зарегистрирована", id);
                }

                // 2. ОБНОВЛЕНИЕ КОНКРЕТНОГО ПОЛЯ
                SystemMessage::UpdateEntityField { entity_id, field, source } => {
                    if let Some(entity) = self.entities.get_mut(&entity_id) {
                        // use crate::state::{EntityState, FieldUpdate::*};
                        
                        // Вычисляем, изменилось ли значение на самом деле (защита от дедупликации)
                        let changed = match (&mut entity.state, field) {
                            (EntityState::Light(light), FieldUpdate::IsOn(v)) => {
                                let old = light.is_on; light.is_on = v; old != v
                            }
                            (EntityState::Light(light), FieldUpdate::Brightness(v)) => {
                                let old = light.brightness; light.brightness = v; old != v
                            }
                            (EntityState::Light(light), FieldUpdate::ColorTemp(v)) => {
                                let old = light.color_temp; light.color_temp = v; old != v
                            }
                            (EntityState::Sensor(sensor), FieldUpdate::Value(v)) => {
                                let old = sensor.value; sensor.value = v; old != v
                            }
                            (EntityState::BinarySensor(binary), FieldUpdate::IsOn(v)) => {
                                let old = binary.is_on; binary.is_on = v; old != v
                            }
                            (EntityState::InputBoolean(input_bool), FieldUpdate::IsOn(v)) => {
                                let old = input_bool.is_on; input_bool.is_on = v; old != v
                            }
                            _ => {
                                eprintln!("[Ядро] ⚠️ Несовместимое поле для '{}'", entity_id);
                                false
                            }
                        };

                        // Если значение НЕ изменилось — срабатывает дедупликация и переходим к следующему сообщению.
                        if !changed {
                            // Значение то же самое, защитили систему от лишней работы!
                            continue; 
                        }

                        // Сюда код дойдет ТОЛЬКО если значение реально обновилось
                        println!("[Ядро] 🔄 '{}' обновлена (source: {:?})", entity_id, source);
                        
                        // Отправляем "будильник" в Роутер для пробуждения Lua-скриптов
                        let event = SystemMessage::EntityChangedEvent {
                            entity_id: entity_id.clone(),
                        };
                        let _ = self.tx_to_router.send(event);

                    } else {
                        // Срабатывает, только если id сущности вообще нет в базе данных ядра!
                        eprintln!("[Ядро] ⚠️ UpdateEntityField: неизвестная сущность '{}'", entity_id);
                    }
                }

                // 3. ОБРАБОТКА ЗАПРОСА СОСТОЯНИЯ 
                SystemMessage::GetEntityState { entity_id, reply } => {
                    let entity_clone = self.entities.get(&entity_id).cloned();
                    // Отправляем ответ НАПРЯМУЮ в вложенный конверт (минуя Роутер)
                    let _ = reply.send(entity_clone);
                }

                // 4. ПЛАВНОЕ ЗАВЕРШЕНИЕ
                SystemMessage::Shutdown => {
                    println!("[Ядро] Получен сигнал Shutdown. Остановка...");
                    break;
                }

                // Все остальные сообщения (например, события для Lua) Ядро просто игнорирует
                _ => {}
            }
        }
        println!("[Ядро] Менеджер сущностей успешно остановлен.");
    }
}

use ent_types::EntityState;
use mlua::UserData;
use std::collections::HashMap;



#[derive(Debug, Clone, PartialEq, UserData)]
pub enum StateKind {
    Light,
    Sensor,
    BinarySensor,
    Counter,
    InputBoolean,
}

impl StateKind {
    pub fn from_state(state: &EntityState) -> Self {
        match state {
            EntityState::Light(_) => StateKind::Light,
            EntityState::Sensor(_) => StateKind::Sensor,
            EntityState::BinarySensor(_) => StateKind::BinarySensor,
            EntityState::Counter(_) => StateKind::Counter,
            EntityState::InputBoolean(_) => StateKind::InputBoolean,
        }
    }
}

pub struct StateFiledKind {
    pub map: HashMap<String, StateKind>,
}

impl StateFiledKind {
    pub fn new() -> Self {
        Self {map: HashMap::new() }
    }
}

use std::println;
use std::thread;
use tokio::sync::mpsc;

use cfg::Config;
use ent_types::SystemMessage;
use ent_router::Router;
use ent_core::EntityManager;
use ent_core::handle::EntityHandle;
use ent_lua::runner::ScriptRunner;
use ent_mqtt::{MqttBridge, mqtt_entities_from_config};

#[tokio::main]
async fn main() {
    println!("=== [entcore] Запуск системы умного дома (Топология Звезда) ===");

    // 1. Читаем конфиг (твоя оригинальная логика)
    let config = Config::load_config("config.toml").expect("Не удалось загрузить конфиг");
    println!("✅ Конфиг загружен");

    // 2. Создаем каналы обратной связи (рельсы ОТ Роутера К модулям)
    let (tx_to_core, rx_for_core) = mpsc::unbounded_channel::<SystemMessage>();
    let (tx_to_lua, rx_for_lua) = mpsc::unbounded_channel::<SystemMessage>();
    let (tx_to_mqtt, _rx_for_mqtt) = mpsc::unbounded_channel::<SystemMessage>();

    // 3. Инициализируем Роутер
    // Получаем сам роутер и входной tx_in, через который все модули будут слать сообщения
    let (router, tx_in) = Router::new(tx_to_core, tx_to_lua, tx_to_mqtt);

    // 4. Выделяем под Роутер отдельный поток операционной системы
    thread::Builder::new()
        .name("[entcore] system-router".into())
        .spawn(move || {
            let rt = tokio::runtime::Builder::new_current_thread()
                .enable_all()
                .build()
                .expect("Не удалось запустить рантайм для Роутера");

            rt.block_on(router.run());
        })
        .expect("Не удалось запустить поток Роутера");

    println!("✅ Поток центрального Роутера запущен");

    // 5. Запускаем EntityManager (Ядро)
    // Оно будет слушать свою ветку rx_for_core и слать ответы/события обратно в Роутер через tx_in
    EntityManager::spawn(rx_for_core, tx_in.clone());
    println!("✅ EntityManager (Ядро) успешно запущен.");

    // 6. Конвертируем MQTT-сущности из конфига и регистрируем их в Системе
    let mqtt_entities = mqtt_entities_from_config(&config.mqtt);
    println!("📦 Обнаружено {} MQTT-сущностей в конфиге. Регистрируем...", mqtt_entities.len());
    for entity in mqtt_entities {
        tx_in.send(SystemMessage::RegisterEntity(entity))
            .expect("Не удалось отправить команду регистрации сущности");
    }

    // 7. Создаем EntityHandle для Lua, который смотрит на Роутер
    let lua_handle = EntityHandle::new(tx_in.clone());

    // 8. Запуск движка автоматизации Lua ScriptRunner
    ScriptRunner::spawn(
        config.system.scripts_directory.clone(),
        rx_for_lua,
        lua_handle,
    );
    println!("✅ Движок автоматизации ScriptRunner (Lua) запущен.");

    // 9. Запуск MQTT-моста
    let integrations = &config.integrations;
    let mqtt_config = integrations.mqtt.as_ref().expect("Секция [integrations.mqtt] обязательна");

    // Передаем мосту клон tx_in, чтобы он слал FieldUpdate сообщения на Роутер
    let mqtt_bridge = MqttBridge::new(mqtt_config, &config.mqtt, tx_in.clone());

    tokio::spawn(async move {
        mqtt_bridge.run().await;
    });
    println!("✅ Мост MQTT запущен и подключается к брокеру.");



    // 10. Основной рабочий цикл ctl-модуля (MVP работает 10 секунд и изящно завершается)
    println!("\n📡 Система полностью активна и обрабатывает события. Ждем 10 секунд...");
    tokio::time::sleep(std::time::Duration::from_secs(10)).await;

    println!("\n🛑 Инициировано плановое завершение работы системы...");
    let _ = tx_in.send(SystemMessage::Shutdown);
    
    // Микро-пауза, чтобы потоки успели напечатать логи закрытия
    tokio::time::sleep(std::time::Duration::from_millis(200)).await;
    println!("=== [entcore] Работа завершена ===");
}

// ./crates/ent_lua/src/runner.rs
use crate::ScriptEngine;
use crate::script_engine::AutomationEvent;
use ent_core::handle::EntityHandle;
use ent_types::SystemMessage; // Работаем с единым типом сообщений
use tokio::sync::mpsc::UnboundedReceiver;
use std::path::Path;

pub struct ScriptRunner;

impl ScriptRunner {
    /// Спавнит ScriptRunner в отдельном потоке ОС.
    /// Он принимает rx_for_lua (рельсы от Роутера) и EntityHandle (для get_state из Lua).
    pub fn spawn(
        scripts_dir: impl AsRef<Path> + Send + 'static,
        mut rx_for_lua: UnboundedReceiver<SystemMessage>,
        tx_handle: EntityHandle,
    ) {
        std::thread::Builder::new()
            .name("[entcore] script-runner".into())
            .spawn(move || {
                // Создаем локальный рантайм для чтения из асинхронного канала rx_for_lua
                let rt = tokio::runtime::Builder::new_current_thread()
                    .enable_all()
                    .build()
                    .expect("Не удалось запустить рантайм для ScriptRunner");

                rt.block_on(async {
                    let (mut engine, receiver) = ScriptEngine::new(tx_handle)
                        .expect("Ошибка инициализации ScriptEngine");
                    
                    engine.register_log().unwrap();
                    engine.registr_entity_id().unwrap();
                    engine.get_state().unwrap();
                    
                    engine.load_scripts(scripts_dir, receiver)
                        .expect("Не удалось загрузить Lua скрипты");

                    println!("[Lua] Движок автоматизации запущен и готов к событиям.");

                    // Читаем сообщения, прилетающие от центрального Роутера
                    while let Some(msg) = rx_for_lua.recv().await {
                        match msg {
                            // Роутер будит нас, если какая-то сущность изменилась!
                            SystemMessage::EntityChangedEvent { entity_id } => {
                                // Передаем событие в Lua. Пока передаем 0.0, 
                                // так как Lua сам пойдет в get_state за нужными полями
                                let event = AutomationEvent {
                                    entity_id,
                                    value: 0.0, 
                                };
                                let _ = engine.call_functions(event);
                            }
                            SystemMessage::Shutdown => {
                                println!("[Lua] Получен сигнал Shutdown. Остановка скриптов...");
                                break;
                            }
                            _ => {} // Все остальные сообщения Lua-движок игнорирует
                        }
                    }
                });
            })
            .expect("Не удалось запустить поток ScriptRunner");
    }
}

use crate::source;
use ent_core::state_filed_kind::StateFiledKind;
use ent_core::handle::EntityHandle;
use ent_types::EntityState; 
use mlua::{Function, Lua, UserData};
use std::{
    collections::HashMap,
    path::Path,
    println,
    sync::mpsc::{self, Receiver},
    time::Duration,
};

#[derive(UserData, Clone)]
pub struct AutomationEvent {
    pub entity_id: String,
    pub value: f64,
}

pub struct ScriptEngine {
    pub lua: Lua,
    pub functions: Vec<Function>,
    pub subscriptions: HashMap<String, Vec<usize>>, // entity_id -> Vec<usize>> где usize это номер элемента вектора в  pub functions: Vec<Function>,
    pub tx: mpsc::Sender<Vec<String>>,
    pub tx_handle: EntityHandle,
    pub state_kind: StateFiledKind,
}

impl ScriptEngine {
    pub fn new(
        tx_handle: EntityHandle,
    ) -> Result<(Self, mpsc::Receiver<Vec<String>>), mlua::Error> {
        let lua = Lua::new();
        let (tx, rx) = mpsc::channel();
        let state_kind = StateFiledKind::new();
        Ok((
            Self {
                lua,
                functions: Vec::new(),
                subscriptions: HashMap::new(),
                tx,
                tx_handle,
                state_kind,
            },
            rx,
        ))
    }

    pub fn load_scripts(
        &mut self,
        dir: impl AsRef<Path>,
        rx: Receiver<Vec<String>>,
    ) -> Result<(), Box<dyn std::error::Error>> {
        let scripts: Vec<source::Script> = source::load_scripts(dir)?;
        for script in scripts {
            let func: Function = self
                .lua
                .load(&script.source_code)
                .set_name(&script.name) // ← полезно для ошибок
                .eval()?;
            let num = self.functions.len();
            match rx.recv_timeout(Duration::from_millis(500)) {
                Ok(ids) => {
                    for id in ids {
                        self.subscriptions.entry(id).or_default().push(num);
                    }
                }
                Err(e) => {
                    println!(
                        "Ошибка ({}) вызова функции registr_entity_id в скрипте {}",
                        e, script.name
                    );
                    continue;
                }
            }
            self.functions.push(func);
        }
        Ok(())
    }

    pub fn call_all_functions(&self, event: AutomationEvent) -> Result<(), mlua::Error> {
        for f in &self.functions {
            f.call::<()>(event.clone())?;
        }
        Ok(())
    }
    pub fn call_functions(&self, event: AutomationEvent) -> Result<(), mlua::Error> {
        let functions = self.subscriptions.get(&event.entity_id);
        match functions {
            Some(functions) => {
                for &f in functions {
                    self.functions[f].call::<()>(event.clone())?;
                }
            }
            None => {
                println!("Скрипты не подписаны на сущность {}", &event.entity_id)
            }
        }
        Ok(())
    }

    pub fn register_log(&self) -> Result<(), mlua::Error> {
        let log_fn = self.lua.create_function(|_, msg: String| {
            println!("[Lua] {}", msg);
            Ok(())
        })?;
        self.lua.globals().set("log", log_fn)?;
        Ok(())
    }

    pub fn registr_entity_id(&self) -> Result<(), mlua::Error> {
        let tx = self.tx.clone();
        let f = self.lua.create_function(move |_, ids: Vec<String>| {
            tx.send(ids).unwrap();
            Ok(())
        })?;
        self.lua.globals().set("registr_entity_id", f)?;
        Ok(())
    }

    pub fn get_state(&self) -> Result<(), mlua::Error> {
        let handle = self.tx_handle.clone();
        let lua = self.lua.clone();
        let f = self.lua.create_function(move |_, entity_id: String| {
            match handle.get_state_blocking(&entity_id) {
                Some(e) => match e.state {
                    EntityState::Light(l) => Ok(mlua::Value::UserData(lua.create_userdata(l)?)),
                    EntityState::Sensor(s) => Ok(mlua::Value::Number(s.value)),
                    EntityState::BinarySensor(b) => Ok(mlua::Value::Boolean(b.is_on)),
                    EntityState::InputBoolean(i) => Ok(mlua::Value::Boolean(i.is_on)),
                    EntityState::Counter(c) => Ok(mlua::Value::Number(c.value as f64)),
                },
                None => Ok(mlua::Value::Nil),
            }
        })?;
        self.lua.globals().set("get_state", f)?;
        Ok(())
    }

    // pub set_state
}

use std::println;
use std::time::Duration;
use tokio::sync::mpsc::UnboundedSender;
use rumqttc::{AsyncClient, Event, Incoming, MqttOptions, QoS};

use cfg::{Mqtt, MqttBroker};
use crate::converter::{TopicKind, Topics, mqtt_from_config};

use ent_types::{SystemMessage, FieldUpdate, UpdateSource};

pub struct MqttBridge {
    client: AsyncClient,
    eventloop: rumqttc::EventLoop,
    topics: Topics,
    tx_to_router: UnboundedSender<SystemMessage>,
}

impl MqttBridge {
    pub fn new(broker: &MqttBroker, mqtt: &Mqtt, tx_to_router: UnboundedSender<SystemMessage>) -> Self {
        let mut options = MqttOptions::new("entcore", &broker.host, broker.port);
        options.set_keep_alive(Duration::from_secs(5));

        let (client, eventloop) = AsyncClient::new(options, 50);

        let topics = mqtt_from_config(mqtt);

        Self {
            client,
            eventloop,
            topics,
            tx_to_router,
        }
    }

    pub async fn run(mut self) {
        let mut subscribe: Vec<String> = Vec::new();
        loop {
            match self.eventloop.poll().await {
                Ok(Event::Incoming(Incoming::Publish(p))) => {
                    let payload = String::from_utf8_lossy(&p.payload);
                    if let Some(mapping) = self.topics.state_topic.get(&p.topic) {
                        // Парсим сырые данные из MQTT в наши типизированные FieldUpdate
                        let field = match mapping.kind {
                            TopicKind::LightIsOn => {
                                let v = payload == "1" || payload == "true" || payload == "ON";
                                FieldUpdate::IsOn(v)
                            }
                            TopicKind::LightBrightness => {
                                FieldUpdate::Brightness(payload.parse().ok())
                            }
                            TopicKind::LightColorTemp => {
                                FieldUpdate::ColorTemp(payload.parse().ok())
                            }
                            TopicKind::SensorValue => {
                                FieldUpdate::Value(payload.parse().unwrap_or(0.0))
                            }
                        };
                        // Создаем сообщение Роутеру!
                        let msg = SystemMessage::UpdateEntityField {
                            entity_id: mapping.entity_id.clone(),
                            field,
                            source: UpdateSource::Mqtt,
                        };

                        // Отправляем на центральный Роутер. Если роутер упал, .is_err() это отловит.
                        if self.tx_to_router.send(msg).is_err() {
                            println!("⚠️ [MQTT Мост] Центральный роутер остановился. Выходим.");
                            break;
                        }

                    }
                }
                Ok(Event::Incoming(Incoming::ConnAck(_))) => {
                    println!("✅ MQTT: ConnAck подключено к брокеру");
                    subscribe.clear();
                    for topic in self.topics.state_topic.keys() {
                        let t = topic.to_string();
                        println!("Топик {} добавлен в вектор", &t);
                        subscribe.push(t);
                    }
                    if let Some(x) = subscribe.pop() {
                        println!("Подписываемся на топик {}", &x);
                        let _ = self.client.subscribe(x, QoS::AtLeastOnce).await;
                    }
                }
                Ok(Event::Incoming(Incoming::SubAck(s))) => {
                    println!("✅ MQTT: SubAck подтверждение от брокера {:?}", s);
                    if let Some(x) = subscribe.pop() {
                        println!("Подписываемся на топик {}", &x);
                        let _ = self.client.subscribe(x, QoS::AtLeastOnce).await;
                    }
                }
                Ok(_) => {}
                Err(e) => {
                    eprintln!("⚠️ MQTT ошибка: {}", e);
                    // при ошибке — пауза, чтобы не спамить
                    tokio::time::sleep(Duration::from_secs(1)).await;
                }
            }
        }
    }
}

use cfg::Mqtt;
use ent_types::{Entity, EntityState, LightState, SensorState};
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub enum TopicKind {
    LightIsOn,
    LightBrightness,
    LightColorTemp,
    SensorValue,
}

#[derive(Debug, Clone)]
pub struct FieldMapping {
    pub entity_id: String,
    pub kind: TopicKind,
}
pub struct Topics {
    pub state_topic: HashMap<String, FieldMapping>, // topic → mapping
}

//-> Vec<Entity>
pub fn mqtt_entities_from_config(mqtt: &Mqtt) -> Vec<Entity> {
    let mut entities: Vec<Entity> = Vec::new();

    for item in &mqtt.light {
        let entity = Entity {
            id: item.entity_id.clone(),
            friendly_name: item.name.clone(),
            state: EntityState::Light(LightState {
                is_on: false,
                brightness: None,
                color_temp: None,
            }),
        };
        entities.push(entity);
    }

    for item in &mqtt.sensor {
        let entity = Entity {
            id: item.entity_id.clone(),
            friendly_name: item.name.clone(),
            state: EntityState::Sensor(SensorState { value: 0.0 }),
        };
        entities.push(entity);
    }

    entities
}

pub fn mqtt_from_config(mqtt: &Mqtt) -> Topics {
    let mut topics = Topics {
        state_topic: HashMap::new(),
    };

    for light in &mqtt.light {
        topics.state_topic.insert(
            light.state_topic.clone(),
            FieldMapping {
                entity_id: light.entity_id.clone(),
                kind: TopicKind::LightIsOn,
            },
        );
        if let Some(topic) = &light.brightness_state_topic {
            topics.state_topic.insert(
                topic.clone(),
                FieldMapping {
                    entity_id: light.entity_id.clone(),
                    kind: TopicKind::LightBrightness,
                },
            );
        }
        if let Some(topic) = &light.color_temp_state_topic {
            topics.state_topic.insert(
                topic.clone(),
                FieldMapping {
                    entity_id: light.entity_id.clone(),
                    kind: TopicKind::LightColorTemp,
                },
            );
        }
    }

    for sensor in &mqtt.sensor {
        topics.state_topic.insert(
            sensor.state_topic.clone(),
            FieldMapping {
                entity_id: sensor.entity_id.clone(),
                kind: TopicKind::SensorValue,
            },
        );
    }

    topics
}

// ./crates/ent_router/src/lib.rs
use ent_types::SystemMessage;
use tokio::sync::mpsc::{UnboundedReceiver, UnboundedSender};

pub struct Router {
    rx_in: UnboundedReceiver<SystemMessage>,
    _tx_in: UnboundedSender<SystemMessage>, // добавили _
    tx_to_core: UnboundedSender<SystemMessage>,
    tx_to_lua: UnboundedSender<SystemMessage>,
    _tx_to_mqtt: UnboundedSender<SystemMessage>, // добавили _
}

impl Router {
    pub fn new(
        tx_to_core: UnboundedSender<SystemMessage>,
        tx_to_lua: UnboundedSender<SystemMessage>,
        tx_to_mqtt: UnboundedSender<SystemMessage>,
    ) -> (Self, UnboundedSender<SystemMessage>) {
        let (tx_in, rx_in) = tokio::sync::mpsc::unbounded_channel();

        let router = Self {
            rx_in,
            _tx_in: tx_in.clone(),
            tx_to_core,
            tx_to_lua,
            _tx_to_mqtt: tx_to_mqtt,
        };

        (router, tx_in)
    }

    pub async fn run(mut self) {
        println!("[Роутер] Центральный маршрутизатор запущен.");
        
        while let Some(msg) = self.rx_in.recv().await {
            match msg {
                SystemMessage::RegisterEntity(_) 
                | SystemMessage::UpdateEntityField { .. } 
                | SystemMessage::GetEntityState { .. } => { 
                    let _ = self.tx_to_core.send(msg);
                }
                SystemMessage::EntityChangedEvent { .. } => {
                    let _ = self.tx_to_lua.send(msg);
                }
                SystemMessage::Shutdown => {
                    println!("[Роутер] Получен сигнал Shutdown. Останавливаем модули...");
                    let _ = self.tx_to_core.send(SystemMessage::Shutdown);
                    let _ = self.tx_to_lua.send(SystemMessage::Shutdown);
                    break;
                }
            }
        }
        println!("[Роутер] Маршрутизатор успешно остановлен.");
    }
}

// ./crates/ent_types/src/lib.rs
use mlua::UserData;
use tokio::sync::oneshot;

// =====================================================================
// 1. ИСТОЧНИКИ ОБНОВЛЕНИЙ И СТРУКТУРЫ СОСТОЯНИЙ
// =====================================================================

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum UpdateSource {
    Mqtt,
    Script,
    Timer,
    Helper,
    System,
}

#[derive(Debug, Clone, PartialEq)]
pub enum EntityState {
    Light(LightState),
    Sensor(SensorState),
    BinarySensor(BinarySensorState),
    Counter(CounterState),
    InputBoolean(InputBooleanState),
}

#[derive(Debug, Clone, PartialEq, UserData)]
pub struct LightState {
    pub is_on: bool,
    pub brightness: Option<u8>,
    pub color_temp: Option<u16>,
}

#[derive(Debug, Clone, PartialEq)]
pub struct SensorState {
    pub value: f64,
}

#[derive(Debug, Clone, PartialEq)]
pub struct BinarySensorState {
    pub is_on: bool,
}

#[derive(Debug, Clone, PartialEq)]
pub struct CounterState {
    pub value: u64,
}

#[derive(Debug, Clone, PartialEq)]
pub struct InputBooleanState {
    pub is_on: bool,
}

#[derive(Debug, Clone, PartialEq)]
pub enum FieldUpdate {
    IsOn(bool),
    Brightness(Option<u8>),
    ColorTemp(Option<u16>),
    Value(f64),
}

// =====================================================================
// 2. СУЩНОСТЬ (ENTITY)
// =====================================================================

#[derive(Debug, Clone, PartialEq)]
pub struct Entity {
    pub id: String,
    pub friendly_name: String,
    pub state: EntityState,
}

// =====================================================================
// 3. ЦЕНТРАЛЬНЫЙ enum СООБЩЕНИЙ (SYSTEM MESSAGE)
// =====================================================================

#[derive(Debug)]
pub enum SystemMessage {
    /// Регистрирует новую сущность в базе данных ядра при старте модуля.
    RegisterEntity(Entity),

    /// Обновляет конкретное поле сущности.
    UpdateEntityField {
        entity_id: String,
        field: FieldUpdate,
        source: UpdateSource,
    },

    /// Запрос состояния сущности (например, из Lua) с одноразовым каналом ответа.
    GetEntityState {
        entity_id: String,
        reply: oneshot::Sender<Option<Entity>>,
    },

    /// Сигнализирует движку скриптов, что состояние сущности изменилось.
    EntityChangedEvent {
        entity_id: String,
    },

    /// Команда на плавную остановку всех фоновых потоков и тасок.
    Shutdown,
}
