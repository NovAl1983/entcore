use serde::Deserialize;
use std::error::Error;

#[derive(Debug, Deserialize, Clone)]
pub struct Config {
    pub system: System,
    pub integrations: Integrations,
    #[serde(default)]
    pub mqtt: Mqtt,
    #[serde(default)]
    pub http: Vec<Http>,
    #[serde(default)]
    pub helper: Helper,
}

impl Config {
    pub fn load_config(path: &str) -> Result<Self, Box<dyn Error + Send + Sync>> {
        let cfg_str = std::fs::read_to_string(path)?;
        let config: Config = toml::from_str(&cfg_str)?;

        let has_integration =
            config.integrations.mqtt.is_some() || config.integrations.modbus_rtu.is_some();
        let has_helper = !config.helper.input_boolean.is_empty();

        if !has_integration && !has_helper {
            return Err(format!(
                "Конфиг {path} пуст: нет ни интеграций (mqtt/modbus_rtu), \
                 ни helpers (input_boolean). Движку не с чем работать."
            )
            .into());
        }

        Ok(config)
    }
}

use tokio::sync::mpsc;
use tokio::sync::oneshot;

use crate::entity::Entity;
use crate::message::{EntityMessage, Notification, UpdateSource};
use crate::state::{EntityState, FieldUpdate};

/// Пульт управления EntityManager.
/// Клонируется, раздаётся модулям.
#[derive(Clone)]
pub struct EntityHandle {
    tx: mpsc::UnboundedSender<EntityMessage>,
}

impl EntityHandle {
    pub(crate) fn new(tx: mpsc::UnboundedSender<EntityMessage>) -> Self {
        Self { tx }
    }

    /// Зарегистрировать сущность. Fire-and-forget.
    pub fn register(&self, entity: Entity) {
        let _ = self.tx.send(EntityMessage::Register(entity));
    }

    /// Обновить состояние. Fire-and-forget.
    pub fn update_state(
        &self,
        entity_id: impl Into<String>,
        new_state: EntityState,
        source: UpdateSource,
    ) {
        let _ = self.tx.send(EntityMessage::UpdateState {
            entity_id: entity_id.into(),
            new_state,
            source,
        });
    }
    /// Обновить состояние конкретного поля сущности
    pub fn update_field(
        &self,
        entity_id: impl Into<String>,
        field: FieldUpdate,
        source: UpdateSource,
    ) {
        let _ = self.tx.send(EntityMessage::UpdateStateField {
            entity_id: entity_id.into(),
            field,
            source,
        });
    }

    // блокирующий метод для сихронного кода
    pub fn get_state_blocking(&self, entity_id: &str) -> Option<Entity> {
        let (tx, rx) = oneshot::channel();
        self.tx
            .send(EntityMessage::GetState {
                entity_id: entity_id.to_string(),
                reply: tx,
            })
            .ok()?;
        rx.blocking_recv().ok()?
    }

    // не блокирующий метод для async
    pub async fn get_state(&self, entity_id: &str) -> Option<Entity> {
        let (tx, rx) = oneshot::channel();
        self.tx
            .send(EntityMessage::GetState {
                entity_id: entity_id.to_string(),
                reply: tx,
            })
            .ok()?;
        rx.await.ok()?
    }

    /// Подождать, пока менеджер обработает все текущие сообщения в очереди.
    /// Этот метод асинхронный, его можно вызывать в async-контексте через .await
    pub async fn wait_ready(&self) -> Result<(), String> {
        let (tx, rx) = oneshot::channel();

        // Отправляем сообщение в очередь менеджера
        if self
            .tx
            .send(EntityMessage::WaitReady { reply: tx })
            .is_err()
        {
            return Err("EntityManager уже остановлен".to_string());
        }

        // Ждем, пока менеджер обработает все сообщения перед нами и ответит в rx
        rx.await
            .map_err(|_| "Поток EntityManager аварийно завершился".to_string())
    }

    pub fn subscribe_blocking(
        &self,
    ) -> Option<(
        Vec<Entity>,
        tokio::sync::mpsc::UnboundedReceiver<Notification>,
    )> {
        let (tx, rx) = tokio::sync::oneshot::channel();
        self.tx.send(EntityMessage::Subscribe { reply: tx }).ok()?;
        rx.blocking_recv().ok()
    }
}

use std::collections::HashMap;
use std::{println, thread};

use tokio::sync::mpsc;

use crate::entity::Entity;
use crate::handle::EntityHandle;
use crate::message::{EntityMessage, Notification};
use crate::state::{EntityState, FieldUpdate};
use crate::state_filed_kind::StateKind;

pub struct EntityManager {
    rx: mpsc::UnboundedReceiver<EntityMessage>,
    pub entities: HashMap<String, Entity>,
    subscribers: Vec<tokio::sync::mpsc::UnboundedSender<Notification>>,
}

impl EntityManager {
    pub fn spawn() -> EntityHandle {
        let (tx, rx) = mpsc::unbounded_channel();

        let manager = Self {
            rx,
            entities: HashMap::new(),
            subscribers: Vec::new(),
        };

        thread::Builder::new()
            .name("entity-manager".into())
            .spawn(move || manager.run())
            .expect("Не удалось запустить поток EntityManager");

        EntityHandle::new(tx)
    }

    fn run(mut self) {
        while let Some(msg) = self.rx.blocking_recv() {
            match msg {
                EntityMessage::Register(entity) => {
                    let id = entity.id.clone();
                    if self.entities.contains_key(&id) {
                        println!("Сущность с  id = {} уже существует", id);
                        continue;
                    }

                    let notification = Notification::Registered {
                        id: entity.id.clone(),
                        kind: StateKind::from_state(&entity.state),
                    };

                    for tx in &self.subscribers {
                        let _ = tx.send(notification.clone());
                    }

                    self.entities.insert(entity.id.clone(), entity);
                    println!("Сущность с  id = {} добавленна в entities", id);
                }
                EntityMessage::WaitReady { reply } => {
                    println!(
                        "📥 EntityManager: получен запрос WaitReady, подтверждаем готовность..."
                    );
                    // Отправляем сигнал обратно. Если вызывающий поток уже умер, игнорируем ошибку (.ok())
                    let _ = reply.send(());
                }
                EntityMessage::UpdateState {
                    entity_id,
                    new_state,
                    source,
                } => {
                    if let Some(entity) = self.entities.get_mut(&entity_id) {
                        if entity.state == new_state {
                            continue;
                        }
                        entity.state = new_state;
                        println!("🔄 '{}' обновлена (source: {:?})", entity_id, source);
                    } else {
                        eprintln!("⚠️ UpdateState: неизвестная '{}'", entity_id);
                    }
                }
                EntityMessage::UpdateStateField {
                    entity_id,
                    field,
                    source,
                } => {
                    if let Some(entity) = self.entities.get_mut(&entity_id) {
                        match (&mut entity.state, field) {
                            (EntityState::Light(light), FieldUpdate::IsOn(v)) => {
                                if light.is_on == v {
                                    continue;
                                }
                                light.is_on = v;
                            }
                            (EntityState::Light(light), FieldUpdate::Brightness(v)) => {
                                if light.brightness == v {
                                    continue;
                                }
                                light.brightness = v;
                            }
                            (EntityState::Light(light), FieldUpdate::ColorTemp(v)) => {
                                if light.color_temp == v {
                                    continue;
                                }
                                light.color_temp = v;
                            }
                            (EntityState::Sensor(sensor), FieldUpdate::Value(v)) => {
                                if sensor.value == v {
                                    continue;
                                }
                                sensor.value = v;
                            }
                            (EntityState::BinarySensor(binary), FieldUpdate::IsOn(v)) => {
                                if binary.is_on == v {
                                    continue;
                                }
                                binary.is_on = v;
                            }
                            (EntityState::InputBoolean(input_bool), FieldUpdate::IsOn(v)) => {
                                if input_bool.is_on == v {
                                    continue;
                                }
                                input_bool.is_on = v;
                            }
                            _ => {
                                eprintln!("⚠️ Несовместимое поле для '{}'", entity_id);
                                continue;
                            }
                        }

                        let notification = Notification::Changed {
                            id: entity_id.clone(),
                            state: entity.state.clone(),
                        };
                        for tx in &self.subscribers {
                            let _ = tx.send(notification.clone());
                        }

                        println!("🔄 '{}' обновлена (field, source: {:?})", entity_id, source);
                    } else {
                        eprintln!("⚠️ UpdateStateField: неизвестная '{}'", entity_id);
                    }
                }
                EntityMessage::GetState { entity_id, reply } => {
                    let entity = self.entities.get(&entity_id).cloned();
                    let _ = reply.send(entity);
                }
                EntityMessage::ListAll { reply } => {
                    let all: Vec<Entity> = self.entities.values().cloned().collect();
                    let _ = reply.send(all);
                }
                EntityMessage::Shutdown => {
                    println!("EntityManager: получен Shutdown");
                    break;
                }
                EntityMessage::Subscribe { reply } => {
                    let (tx, rx) = tokio::sync::mpsc::unbounded_channel();
                    self.subscribers.push(tx);
                    let snapshot: Vec<Entity> = self.entities.values().cloned().collect();
                    let _ = reply.send((snapshot, rx));
                }
            }
        }
    }

    pub fn get_entities(self) -> HashMap<String, Entity> {
        let e = self.entities.clone();
        e
    }
}

use crate::entity::Entity;
use crate::state::{EntityState, FieldUpdate};
use crate::state_filed_kind::StateKind;
use tokio::sync::oneshot;
// use tokio::sync::mpsc;
use std::sync::mpsc;

/// Сообщения, которые модули шлют в EntityManager.
#[derive(Debug)]
pub enum EntityMessage {
    /// Зарегистрировать сущность при старте.
    Register(Entity),

    /// Новый вариант: подождать, пока менеджер обработает все предыдущие сообщения.
    /// Ответ (сигнал готовности) вернется через oneshot-канал.
    WaitReady { reply: oneshot::Sender<()> },

    /// Обновить состояние сущности.
    UpdateState {
        entity_id: String,
        new_state: EntityState,
        source: UpdateSource,
    },

    UpdateStateField {
        entity_id: String,
        field: FieldUpdate,
        source: UpdateSource,
    },

    /// Прочитать сущность по id. Ответ — через oneshot.
    GetState {
        entity_id: String,
        reply: oneshot::Sender<Option<Entity>>,
    },

    /// Получить список всех сущностей. Для отладки и WebUI.
    ListAll { reply: oneshot::Sender<Vec<Entity>> },

    Subscribe {
        reply: oneshot::Sender<(
            Vec<Entity>,
            tokio::sync::mpsc::UnboundedReceiver<Notification>,
        )>,
    },

    /// Завершить работу актора.
    Shutdown,
}

/// Кто инициировал изменение.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum UpdateSource {
    Mqtt,
    Script,
    Timer,
    Helper,
    System,
}

#[derive(Debug, Clone)]
pub enum Notification {
    Registered { id: String, kind: StateKind },
    Changed { id: String, state: EntityState },
}

pub enum ScriptRunnerMessage {
    EntityChanged(AutomationEvent),
    StateKind((String, StateKind)),
    Shutdown,
}

pub struct ScriptRunnerHandle {
    tx: mpsc::Sender<ScriptRunnerMessage>,
}

impl ScriptRunnerHandle {
    pub fn notify_changed(&self, event: AutomationEvent) {
        let _ = self.tx.send(ScriptRunnerMessage::EntityChanged(event));
    }
    pub fn shutdown(&self) {
        let _ = self.tx.send(ScriptRunnerMessage::Shutdown);
    }
}

pub struct ScriptRunner;

impl ScriptRunner {
    pub fn spawn(
        scripts_dir: impl AsRef<Path> + Send + 'static,
        tx_handle: EntityHandle,
    ) -> ScriptRunnerHandle {
        let (tx, rx) = mpsc::channel();

        std::thread::Builder::new()
            .name("script-runner".into())
            .spawn(move || {
                // let mut engine = ScriptEngine::new().expect(r"ошибка в crates\ent_lua\src\runner.rs - fn spawn");
                let (mut engine, receiver) = ScriptEngine::new(tx_handle)
                    .expect(r"ошибка в crates\ent_lua\src\runner.rs - fn spawn");
                engine
                    .register_log()
                    .expect(r"ошибка в crates\ent_lua\src\runner.rs - fn spawn");
                engine
                    .registr_entity_id()
                    .expect(r"ошибка в crates\ent_lua\src\runner.rs - fn spawn");
                engine
                    .get_state()
                    .expect(r"ошибка в crates\ent_lua\src\runner.rs - fn spawn");
                engine
                    .load_scripts(scripts_dir, receiver)
                    .expect(r"ошибка в crates\ent_lua\src\runner.rs - fn spawn");

                loop {
                    match rx.recv() {
                        Ok(ScriptRunnerMessage::EntityChanged(event)) => {
                            let _ = engine.call_functions(event);
                        }
                        Ok(ScriptRunnerMessage::StateKind((k, v))) => {
                            engine.state_kind.map.insert(k, v);
                        }
                        Ok(ScriptRunnerMessage::Shutdown) => break,
                        Err(e) => {
                            println!("Канал закрыт: {}", e);
                            break;
                        }
                    }
                }
            })
            .expect(r"ошибка в crates\ent_lua\src\runner.rs - fn spawn");
        ScriptRunnerHandle { tx }
    }
}

use crate::source;
use ent_core::EntityManager;
use ent_core::state::SensorState;
use ent_core::state_filed_kind::StateFiledKind;
use ent_core::{EntityState, LightState, handle::EntityHandle};
use mlua::{Function, Lua, UserData};
use std::{
    collections::HashMap,
    path::Path,
    println,
    sync::mpsc::{self, Receiver},
    time::Duration,
};

#[derive(UserData, Clone)]
pub struct AutomationEvent {
    pub entity_id: String,
    pub value: f64,
}

pub struct ScriptEngine {
    pub lua: Lua,
    pub functions: Vec<Function>,
    pub subscriptions: HashMap<String, Vec<usize>>, // entity_id -> Vec<usize>> где usize это номер элемента вектора в  pub functions: Vec<Function>,
    pub tx: mpsc::Sender<Vec<String>>,
    pub tx_handle: EntityHandle,
    pub state_kind: StateFiledKind,
}

impl ScriptEngine {
    pub fn new(
        tx_handle: EntityHandle,
    ) -> Result<(Self, mpsc::Receiver<Vec<String>>), mlua::Error> {
        let lua = Lua::new();
        let (tx, rx) = mpsc::channel();
        let state_kind = StateFiledKind::new();
        Ok((
            Self {
                lua,
                functions: Vec::new(),
                subscriptions: HashMap::new(),
                tx,
                tx_handle,
                state_kind,
            },
            rx,
        ))
    }

    pub fn load_scripts(
        &mut self,
        dir: impl AsRef<Path>,
        rx: Receiver<Vec<String>>,
    ) -> Result<(), Box<dyn std::error::Error>> {
        let scripts: Vec<source::Script> = source::load_scripts(dir)?;
        for script in scripts {
            let func: Function = self
                .lua
                .load(&script.source_code)
                .set_name(&script.name) // ← полезно для ошибок
                .eval()?;
            let num = self.functions.len();
            match rx.recv_timeout(Duration::from_millis(500)) {
                Ok(ids) => {
                    for id in ids {
                        self.subscriptions.entry(id).or_default().push(num);
                    }
                }
                Err(e) => {
                    println!(
                        "Ошибка ({}) вызова функции registr_entity_id в скрипте {}",
                        e, script.name
                    );
                    continue;
                }
            }
            self.functions.push(func);
        }
        Ok(())
    }

    pub fn call_all_functions(&self, event: AutomationEvent) -> Result<(), mlua::Error> {
        for f in &self.functions {
            f.call::<()>(event.clone())?;
        }
        Ok(())
    }
    pub fn call_functions(&self, event: AutomationEvent) -> Result<(), mlua::Error> {
        let functions = self.subscriptions.get(&event.entity_id);
        match functions {
            Some(functions) => {
                for &f in functions {
                    self.functions[f].call::<()>(event.clone())?;
                }
            }
            None => {
                println!("Скрипты не подписаны на сущность {}", &event.entity_id)
            }
        }
        Ok(())
    }

    pub fn register_log(&self) -> Result<(), mlua::Error> {
        let log_fn = self.lua.create_function(|_, msg: String| {
            println!("[Lua] {}", msg);
            Ok(())
        })?;
        self.lua.globals().set("log", log_fn)?;
        Ok(())
    }

    pub fn registr_entity_id(&self) -> Result<(), mlua::Error> {
        let tx = self.tx.clone();
        let f = self.lua.create_function(move |_, ids: Vec<String>| {
            tx.send(ids).unwrap();
            Ok(())
        })?;
        self.lua.globals().set("registr_entity_id", f)?;
        Ok(())
    }

    pub fn get_state(&self) -> Result<(), mlua::Error> {
        let handle = self.tx_handle.clone();
        let lua = self.lua.clone();
        let f = self.lua.create_function(move |_, entity_id: String| {
            match handle.get_state_blocking(&entity_id) {
                Some(e) => match e.state {
                    EntityState::Light(l) => Ok(mlua::Value::UserData(lua.create_userdata(l)?)),
                    EntityState::Sensor(s) => Ok(mlua::Value::Number(s.value)),
                    EntityState::BinarySensor(b) => Ok(mlua::Value::Boolean(b.is_on)),
                    EntityState::InputBoolean(i) => Ok(mlua::Value::Boolean(i.is_on)),
                    EntityState::Counter(c) => Ok(mlua::Value::Number(c.value as f64)),
                },
                None => Ok(mlua::Value::Nil),
            }
        })?;
        self.lua.globals().set("get_state", f)?;
        Ok(())
    }

}

use std::println;
use std::time::Duration;

// use cfg::Mqtt;
use rumqttc::{AsyncClient, Event, Incoming, MqttOptions, QoS};

use ent_core::{EntityHandle, UpdateSource};

use cfg::{Mqtt, MqttBroker};
// use crate::converter::{Topics, mqtt_from_config};

use crate::converter::{TopicKind, Topics, mqtt_from_config};

use ent_core::FieldUpdate;

// use ent_core::UpdateSource;
// use ent_core::EntityHandle;

pub struct MqttBridge {
    client: AsyncClient,
    eventloop: rumqttc::EventLoop,
    topics: Topics,
    // tx: EntityHandle,
    handle: EntityHandle,
}

impl MqttBridge {
    pub fn new(broker: &MqttBroker, mqtt: &Mqtt, handle: EntityHandle) -> Self {
        let mut options = MqttOptions::new("entcore", &broker.host, broker.port);
        options.set_keep_alive(Duration::from_secs(5));

        let (client, eventloop) = AsyncClient::new(options, 50);

        let topics = mqtt_from_config(mqtt);

        Self {
            client,
            eventloop,
            topics,
            handle,
        }
    }

    pub async fn run(mut self) {
        let mut subscribe: Vec<String> = Vec::new();
        loop {
            match self.eventloop.poll().await {
                Ok(Event::Incoming(Incoming::Publish(p))) => {
                    let payload = String::from_utf8_lossy(&p.payload);
                    if let Some(mapping) = self.topics.state_topic.get(&p.topic) {
                        let field = match mapping.kind {
                            TopicKind::LightIsOn => {
                                let v = payload == "1" || payload == "true" || payload == "ON";
                                FieldUpdate::IsOn(v)
                            }
                            TopicKind::LightBrightness => {
                                FieldUpdate::Brightness(payload.parse().ok())
                            }
                            TopicKind::LightColorTemp => {
                                FieldUpdate::ColorTemp(payload.parse().ok())
                            }
                            TopicKind::SensorValue => {
                                FieldUpdate::Value(payload.parse().unwrap_or(0.0))
                            }
                        };
                        self.handle.update_field(
                            mapping.entity_id.clone(),
                            field,
                            UpdateSource::Mqtt,
                        );
                    }
                }
                Ok(Event::Incoming(Incoming::ConnAck(_))) => {
                    println!("✅ MQTT: ConnAck подключено к брокеру");
                    subscribe.clear();
                    for topic in self.topics.state_topic.keys() {
                        let t = topic.to_string();
                        println!("Топик {} добавлен в вектор", &t);
                        subscribe.push(t);
                    }
                    if let Some(x) = subscribe.pop() {
                        println!("Подписываемся на топик {}", &x);
                        let _ = self.client.subscribe(x, QoS::AtLeastOnce).await;
                    }
                }
                Ok(Event::Incoming(Incoming::SubAck(s))) => {
                    println!("✅ MQTT: SubAck подтверждение от брокера {:?}", s);
                    if let Some(x) = subscribe.pop() {
                        println!("Подписываемся на топик {}", &x);
                        let _ = self.client.subscribe(x, QoS::AtLeastOnce).await;
                    }
                }
                Ok(_) => {}
                Err(e) => {
                    eprintln!("⚠️ MQTT ошибка: {}", e);
                    // при ошибке — пауза, чтобы не спамить
                    tokio::time::sleep(Duration::from_secs(1)).await;
                }
            }
        }
    }
}

            }),
        };
        entities.push(entity);
    }

    for item in &mqtt.sensor {
        let entity = Entity {
            id: item.entity_id.clone(),
            friendly_name: item.name.clone(),
            state: EntityState::Sensor(SensorState { value: 0.0 }),
        };
        entities.push(entity);
    }

    entities
}

pub fn mqtt_from_config(mqtt: &Mqtt) -> Topics {
    let mut topics = Topics {
        state_topic: HashMap::new(),
    };

    for light in &mqtt.light {
        topics.state_topic.insert(
            light.state_topic.clone(),
            FieldMapping {
                entity_id: light.entity_id.clone(),
                kind: TopicKind::LightIsOn,
            },
        );
        if let Some(topic) = &light.brightness_state_topic {
            topics.state_topic.insert(
                topic.clone(),
                FieldMapping {
                    entity_id: light.entity_id.clone(),
                    kind: TopicKind::LightBrightness,
                },
            );
        }
        if let Some(topic) = &light.color_temp_state_topic {
            topics.state_topic.insert(
                topic.clone(),
                FieldMapping {
                    entity_id: light.entity_id.clone(),
                    kind: TopicKind::LightColorTemp,
                },
            );
        }
    }

    for sensor in &mqtt.sensor {
        topics.state_topic.insert(
            sensor.state_topic.clone(),
            FieldMapping {
                entity_id: sensor.entity_id.clone(),
                kind: TopicKind::SensorValue,
            },
        );
    }

    topics
}
